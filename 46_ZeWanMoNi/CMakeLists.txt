cmake_minimum_required(VERSION 3.20)
project(ZeWanMoNi LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT NX_ROOT)
  set(NX_ROOT "$ENV{UGII_BASE_DIR}")
endif()
if(NOT EXISTS "${NX_ROOT}/UGOPEN/uf.h")
  message(FATAL_ERROR "Set NX_ROOT to the NX installation directory")
endif()
function(configure_nx target)
  target_include_directories(${target} PRIVATE "${NX_ROOT}/UGOPEN" "${NX_ROOT}/UGOPEN/NXOpen")
  target_compile_definitions(${target} PRIVATE UNICODE _UNICODE NOMINMAX _CRT_SECURE_NO_WARNINGS $<$<CONFIG:Release>:ZH_PROTECTED_BUILD=1>)
  target_compile_options(${target} PRIVATE /W4 /EHsc /utf-8 $<$<CONFIG:Release>:/O2>)
  target_link_directories(${target} PRIVATE "${NX_ROOT}/UGOPEN")
  target_link_libraries(${target} PRIVATE libufun.lib libugopenint.lib libnxopencpp.lib libnxopencpp_gateway.lib libnxopencpp_features.lib libnxopencpp_geometricutilities.lib libnxopenuicpp.lib user32.lib shell32.lib)
  target_link_options(${target} PRIVATE $<$<CONFIG:Release>:/RELEASE> /DYNAMICBASE /NXCOMPAT)
endfunction()
add_library(ZeWanMoNi SHARED ZeWanMoNi.cpp ZeWanMoNiMain.cpp BendSimulation.cpp ToolProfiles.cpp ToolThumbnails.cpp)
configure_nx(ZeWanMoNi)
foreach(resource ZeWanMoNi.dlx ZeWanMoNi.bmp ZeWanMoNiExample.ztool)
  add_custom_command(TARGET ZeWanMoNi POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different "${CMAKE_CURRENT_SOURCE_DIR}/${resource}" "$<TARGET_FILE_DIR:ZeWanMoNi>/${resource}")
endforeach()
add_custom_target(ZeWanMoNiToolLibrary ALL DEPENDS ZeWanMoNi)
foreach(tool 000 001 003 004 RZ8895 RZ30104 RW8867 RW8890 RW88100 RW88120)
  add_custom_command(TARGET ZeWanMoNiToolLibrary POST_BUILD COMMAND ${CMAKE_COMMAND} -E make_directory "$<TARGET_FILE_DIR:ZeWanMoNi>/ZeWanMoNiTools"
    COMMAND ${CMAKE_COMMAND} -E copy_if_different "${CMAKE_CURRENT_SOURCE_DIR}/tool-library/dwg-20260921/tools/DWG_${tool}.ztool" "$<TARGET_FILE_DIR:ZeWanMoNi>/ZeWanMoNiTools/DWG_${tool}.ztool")
endforeach()

if(ZH_BEND_SIM_TESTS)
  add_executable(BendSimulationIntegration tests/Integration.cpp BendSimulation.cpp ToolProfiles.cpp)
  configure_nx(BendSimulationIntegration)

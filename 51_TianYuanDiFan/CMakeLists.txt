cmake_minimum_required(VERSION 3.20)
project(TianYuanDiFan LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT NX_ROOT)
  set(NX_ROOT "$ENV{UGII_BASE_DIR}")
endif()
if(NOT EXISTS "${NX_ROOT}/UGOPEN/uf.h")
  message(FATAL_ERROR "Set NX_ROOT to the NX installation directory")
endif()
function(configure_nx target)
  target_include_directories(${target} PRIVATE "${NX_ROOT}/UGOPEN" "${NX_ROOT}/UGOPEN/NXOpen" "${CMAKE_CURRENT_SOURCE_DIR}/../..")
  target_compile_definitions(${target} PRIVATE UNICODE _UNICODE NOMINMAX _CRT_SECURE_NO_WARNINGS $<$<CONFIG:Release>:ZH_PROTECTED_BUILD=1>)
  target_compile_options(${target} PRIVATE /W4 /EHsc /utf-8 $<$<CONFIG:Release>:/O2>)
  target_link_directories(${target} PRIVATE "${NX_ROOT}/UGOPEN")
  target_link_libraries(${target} PRIVATE libufun.lib libugopenint.lib libnxopencpp.lib libnxopencpp_gateway.lib libnxopencpp_features.lib libnxopencpp_geometricutilities.lib libnxopenuicpp.lib user32.lib)
  target_link_options(${target} PRIVATE $<$<CONFIG:Release>:/RELEASE> /DYNAMICBASE /NXCOMPAT)
endfunction()
add_library(TianYuanDiFan SHARED TianYuanDiFan.cpp TianYuanDiFanMain.cpp TransitionGeometry.cpp)
configure_nx(TianYuanDiFan)
foreach(resource TianYuanDiFan.dlx TianYuanDiFan.bmp)
  add_custom_command(TARGET TianYuanDiFan POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different "${CMAKE_CURRENT_SOURCE_DIR}/${resource}" "$<TARGET_FILE_DIR:TianYuanDiFan>/${resource}")
endforeach()
option(ZH_TRANSITION_TESTS "Build external NX tests; never deploy test executables" OFF)
if(ZH_TRANSITION_TESTS)
  add_executable(TransitionIntegration tests/Integration.cpp TransitionGeometry.cpp)
  configure_nx(TransitionIntegration)
endif()
